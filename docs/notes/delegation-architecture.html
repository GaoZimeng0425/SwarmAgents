┌─────────────────────────────────────────────────────────────────────────────┐ │ Leader Agent (turn run) launchMessage delegate tool session-service message_events │ │ │ │ 1. set_delegation_plan([d1, d2]) │ │ ────────────────────────────────▶ ctx.setDelegationPlan(plan) │ │ ┌─── emit(delegation_plan) ──────────────▶ appendEvent │ │ └─── spec.onDelegationPlan(plan) ──▶ setDelegationPlanForSession │ │ └── planStates.set(sid, ...) │ │ │ │ 2. delegate({ prompt: "...", itemId: "d1" }) │ │ ──────────────────────────────────────────────────────▶ execute starts │ │ ① ctx.mergeDelegationResult("d1", {status:"running"}) │ │ ┌── emit(delegation_update running) ──▶ appendEvent │ │ └── spec.onDelegationUpdate ──▶ mergeDelegationResultForSession │ │ │ │ spawnChild(prompt) ──────▶ delegate(session, ...) │ │ └──▶ launchMessage({kind:"child"}) │ │ │ │ ┌─────────────────── Child Agent (child run) ──────────────────┐ │ │ │ 3. LLM loop with tools + auto-injected report_result │ │ │ │ report_result({ artifacts: [{kind:"note", text:"结果"}] }) │ │ │ │ └─▶ ctx.reportResult(artifacts) │ │ │ │ └─▶ collectedArtifacts.push(...) │ │ │ │ 4. run ends → EngineRunResult { ..., artifacts: collected } │ │ │ └─────────────────────────────────────────────────────────────────┘ │ │ ◀── { messageId, status, summary, artifacts } │ │ │ │ ② ctx.mergeDelegationResult("d1", {status:"completed", artifacts}) │ │ ┌── emit(delegation_update completed) ─▶ appendEvent │ │ └── spec.onDelegationUpdate ──▶ mergeDelegationResultForSession │ │ │ │ ◀── tool_result: { details: { messageId, status, summary, artifacts } } │ │ (Leader 读到结构化 artifacts，继续 loop) │ │ │ │ ✗ 如果 spawnChild 抛异常 │ │ ③ catch: mergeDelegationResult("d1", {status:"failed"}) then rethrow │ │ └── emit(delegation_update failed) ──────▶ appendEvent │ └─────────────────────────────────────────────────────────────────────────────┘
